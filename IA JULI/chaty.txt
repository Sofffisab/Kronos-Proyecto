def convertir_minutos_legibles(minutos):
    if minutos == 30:
        return "30 minutos"
    if minutos == 60:
        return "1 hora"
    if minutos == 90:
        return "1 hora y media"
    if minutos == 120:
        return "2 horas"
    if minutos == 150:
        return "2 horas y media"
    if minutos == 180:
        return "3 horas"
    if minutos == 240:
        return "4 horas"
    return f"{minutos // 60} horas"

def distribuir_tarea_en_dias(descripcion, horario_laboral, fecha_limite, prioridad):
    desc = descripcion.lower()
    if "largo" in desc or "grande" in desc or "extenso" in desc:
        horas_totales = 6
    elif "investigar" in desc:
        horas_totales = 4
    elif "escribir" in desc:
        horas_totales = 3
    elif "corto" in desc or "simple" in desc:
        horas_totales = 1
    else:
        horas_totales = 2

    minutos_totales = horas_totales * 60
    dias_ordenados = ["lunes", "martes", "miércoles", "jueves", "viernes", "sábado", "domingo"]
    fecha_limite_dt = datetime.strptime(fecha_limite, "%Y-%m-%d")
    hoy = datetime.now()

    dias_disponibles = []
    f = hoy
    while f <= fecha_limite_dt:
        nombre = dias_ordenados[f.weekday()]
        if nombre in horario_laboral and horario_laboral[nombre] > 0:
            dias_disponibles.append({
                "fecha": f.strftime("%Y-%m-%d"),
                "minutos": horario_laboral[nombre]
            })
        f += timedelta(days=1)

    plan = []
    peso = 1.5 if prioridad == "alta" else 1 if prioridad == "media" else 0.8

    for dia in dias_disponibles:
        if minutos_totales <= 0:
            break
        asignable = min(int(dia["minutos"] * peso), minutos_totales)
        asignable = (asignable // 30) * 30
        if asignable > 0:
            minutos_totales -= asignable
            plan.append({
                "dia": dia["fecha"],
                "tiempo_asignado": convertir_minutos_legibles(asignable)
            })

    return plan

def calcular_hash_tareas(tareas):
    tareas_json = json.dumps(tareas, sort_keys=True, ensure_ascii=False)
    return hashlib.md5(tareas_json.encode()).hexdigest()

def cargar_hash_anterior():
    if os.path.exists("tareas_hash.txt"):
        with open("tareas_hash.txt", "r", encoding="utf-8") as f:
            return f.read().strip()
    return None

def guardar_hash(hash_nuevo):
    with open("tareas_hash.txt", "w", encoding="utf-8") as f:
        f.write(hash_nuevo)

def cargar_prediccion_guardada():
    if os.path.exists("prediccion_guardada.json"):
        with open("prediccion_guardada.json", "r", encoding="utf-8") as f:
            return json.load(f)
    return None

def guardar_prediccion(pred):
    with open("prediccion_guardada.json", "w", encoding="utf-8") as f:
        json.dump(pred, f, indent=4, ensure_ascii=False)

def _safe_getattr_or_key(obj, name):
    try:
        return getattr(obj, name)
    except:
        try:
            return obj[name]
        except:
            return None

def extract_text_from_response(response):
    try:
        try:
            choice = response.choices[0]
        except:
            choice = response["choices"][0]

        msg = _safe_getattr_or_key(choice, "message")
        content = _safe_getattr_or_key(msg, "content")

        if isinstance(content, str):
            return content.strip()

        if isinstance(content, list) and len(content) > 0:
            block = content[0]
            if isinstance(block, str):
                return block.strip()
            if isinstance(block, dict):
                for k in ("text", "content"):
                    if k in block and isinstance(block[k], str):
                        return block[k].strip()

        txt = _safe_getattr_or_key(choice, "text")
        if isinstance(txt, str):
            return txt.strip()

        return ""
    except Exception as e:
        return f"ERROR_EXTRACTING_TEXT: {str(e)}"

@app.route("/organizar", methods=["POST"])
def organizar_tareas():
    data = request.json
    usuario_logueado = request.headers.get("Usuario")
    if not usuario_logueado:
        return jsonify({"error":"No se encontró usuario logueado"}), 400

    # ⚡ Traer horario laboral real del backend según el usuario logueado
    try:
        # Si la tarea ya tiene horario_laboral, lo usamos
        horario_laboral_usuario = None
        for tarea in data:
            if tarea.get("responsable") == usuario_logueado and tarea.get("horario_laboral"):
                horario_laboral_usuario = tarea["horario_laboral"]
                break
        
        # Si no hay horario en la tarea, intentamos traerlo del backend
        if not horario_laboral_usuario:
            res_horario = requests.get(f"http://localhost:5000/horario/{usuario_logueado}")
            if res_horario.status_code != 200:
                return jsonify({"error":"No se pudo obtener horario laboral"}), 500
            horario_laboral_usuario = res_horario.json()

    except Exception as e:
        return jsonify({"error": f"No se pudo obtener horario laboral: {str(e)}"}), 500

    # Filtrar solo tareas del usuario logueado
    tareas_usuario = []
    for tarea in data:
        if tarea.get("responsable") == usuario_logueado:
            tarea["horario_laboral"] = horario_laboral_usuario
            tareas_usuario.append(tarea)

    # Detectar si cambió
    hash_actual = calcular_hash_tareas(tareas_usuario)
    hash_anterior = cargar_hash_anterior()
    if hash_actual == hash_anterior:
        pred = cargar_prediccion_guardada()
        if pred:
            print("Usando predicción anterior (tareas NO cambiaron)")
            return jsonify({"plan": pred})

    # PROMPT NUEVO
    prompt = f"""
Soy una IA que organiza tareas distribuyéndolas en varios días.
Debo devolver TODOS los días necesarios para completar cada tarea antes de la fecha límite.

Para cada tarea recibo:
- nombre
- responsable
- fecha límite
- prioridad
- estado
- descripción
- horario laboral

Reglas:
- Parto tareas grandes en varios días.
- Respeto horario laboral.
- Indico tiempo diario en formato legible (“1 hora”, “1 hora y media”, “30 minutos”).
- Respondo SOLO JSON válido.
- Sin texto extra.

Formato EXACTO:

[
    {{"nombre": "...", "responsable": "...", "fecha_limite": "...",
      "prioridad": "...", "estado": "...",
      "planificacion":[{{"dia":"YYYY-MM-DD","tiempo_asignado":"1 hora"}}]}}
]

Tareas (datos exactos):
{json.dumps(tareas_usuario, ensure_ascii=False)}
"""

    try:
        response = client.chat.completions.create(
            model="gpt-4o",
            messages=[
                {"role": "system", "content": "Respondé SOLO JSON válido."},
                {"role": "user", "content": prompt}
            ],
            max_tokens=1200,
            temperature=0
        )

        plan_text = extract_text_from_response(response)
        if plan_text.startswith("```"):
            plan_text = plan_text.split("```json")[-1].split("```")[-1].strip()

        try:
            plan_json = json.loads(plan_text)
            guardar_hash(hash_actual)
            guardar_prediccion(plan_json)
            return jsonify({"plan": plan_json})
        except json.JSONDecodeError:
            return jsonify({"plan_text": plan_text})

    except Exception as e:
        return jsonify({"error": str(e)}), 500

@app.route("/calendarizar", methods=["POST"])
def calendarizar_plan():
    plan = request.json
    usuario_logueado = request.headers.get("Usuario")
    if not usuario_logueado:
        return jsonify({"error": "No se encontró usuario logueado"}), 400
    if not plan or len(plan) == 0:
        return jsonify({"error": "No se recibió plan para calendarizar"}), 400

    eventos_google = []
    for tarea in plan:
        horas = 0
        minutos = 0
        tiempo = tarea.get("tiempo_asignado", "")
        if "hora y media" in tiempo:
            horas = 1
            minutos = 30
        elif "hora" in tiempo:
            try:
                horas = int(tiempo.split(" ")[0])
            except:
                horas = 1
        elif "30 minutos" in tiempo:
            minutos = 30

        fecha = tarea.get("dia")
        start = datetime.strptime(fecha, "%Y-%m-%d")
        end = start + timedelta(hours=horas, minutes=minutos)
        eventos_google.append({
            "summary": tarea.get("nombre"),
            "description": f"Tarea: {tarea.get('nombre')}",
            "start": {"dateTime": start.isoformat(), "timeZone": "America/Argentina/Buenos_Aires"},
            "end": {"dateTime": end.isoformat(), "timeZone": "America/Argentina/Buenos_Aires"},
        })

    try:
        res = requests.post(
            "http://localhost:5000/calendar/create",
            json={"usuario": usuario_logueado, "eventos": eventos_google},
            timeout=10
        )
        if res.status_code != 201:
            return jsonify({"error": "No se pudieron crear los eventos, inténtalo de nuevo"}), 500
        return jsonify({"message": "Todos los eventos se calendarizaron correctamente"}), 201
    except Exception as e:
        return jsonify({"error": f"Error al conectar con Google Calendar: {str(e)}"}), 500
